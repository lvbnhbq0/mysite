{{/* Return the authors data map for the current language. 1) База: site.Data.authors (data/authors/*.yaml). 2) Если есть site.Data..authors — поверх накладываем (как в HugoBlox). 3) Для ru: дополнительно читаем data/ru/authors/me.yaml через readFile + transform.Unmarshal и merge в authors.me. Так русские поля подхватываются даже если site.Data.ru пустой (частая ситуация на GitHub Pages / разных версиях Hugo). Usage (как в модуле): {{ $authors := partialCached "functions/get_authors_data" . site.Language.Lang }} */}} {{- $authors := index site.Data "authors" -}} {{- if not $authors -}} {{- $authors = dict -}} {{- end -}} {{- $lang := site.Language.Lang -}} {{- $langData := index site.Data $lang -}} {{- if $langData -}} {{- $langAuthors := index $langData "authors" -}} {{- if $langAuthors -}} {{- $merged := dict -}} {{- range $slug, $data := $authors -}} {{- $langAuthor := index $langAuthors $slug -}} {{- if $langAuthor -}} {{- $mergedAuthor := merge $data $langAuthor -}} {{- $baseName := index $data "name" -}} {{- $langName := index $langAuthor "name" -}} {{- if and (reflect.IsMap $baseName) (reflect.IsMap $langName) -}} {{- $mergedAuthor = merge $mergedAuthor (dict "name" (merge $baseName $langName)) -}} {{- end -}} {{- $merged = merge $merged (dict $slug $mergedAuthor) -}} {{- else -}} {{- $merged = merge $merged (dict $slug $data) -}} {{- end -}} {{- end -}} {{- range $slug, $data := $langAuthors -}} {{- if not (index $authors $slug) -}} {{- $merged = merge $merged (dict $slug $data) -}} {{- end -}} {{- end -}} {{- $authors = $merged -}} {{- end -}} {{- end -}} {{- /* Явный RU-патч из репозитория (не зависит от site.Data.ru) */ -}} {{- if eq $lang "ru" -}} {{- $ruPath := "data/ru/authors/me.yaml" -}} {{- with readFile $ruPath -}} {{- $patch := transform.Unmarshal . -}} {{- if and $patch (reflect.IsMap $patch) -}} {{- $baseMe := index $authors "me" | default dict -}} {{- $mergedMe := merge $baseMe $patch -}} {{- $baseName := index $baseMe "name" -}} {{- $patchName := index $patch "name" -}} {{- if and (reflect.IsMap $baseName) (reflect.IsMap $patchName) -}} {{- $mergedMe = merge $mergedMe (dict "name" (merge $baseName $patchName)) -}} {{- end -}} {{- $authors = merge $authors (dict "me" $mergedMe) -}} {{- end -}} {{- end -}} {{- end -}} {{- return $authors -}}
